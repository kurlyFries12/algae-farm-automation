ORANGE (NØDSITUATION) - DETALJERET SEKVENS
================================================================================

UDLØSNING AF NØDSITUATION:
  Temp < 28 °C ELLER > 37 °C
  ELLER
  pH < 8,5 ELLER > 11,5
  ELLER
  Sensor-fejl/invalid aflæsning
  ELLER
  Pumpe-fejl (flow mangler når det burde være der)

KODE-PSEUDO FÖR NÖDSEKVENTERING:

```cpp
void emergencyShutdown() {
  // 1) START PUMPE BAKUD MED MAX HASTIGHED
  pumpeStepper.setSpeed(PUMP_MAX_SPEED);  // Maksimalt
  pumpeStepper.setDirection(BAKUD);        // DIR = HIGH (bakud)
  
  // 2) SÆT ORANGE LED TIL BLINK
  emergencyLED_state = ORANGE_BLINK_250MS;
  
  // 3) MONITOR FLOW OG VENT PÅ AT DET STOPPER
  unsigned long pumpStartTime = millis();
  unsigned long lastFlowDetectedTime = millis();
  bool flowStoppedDetected = false;
  
  while (!flowStoppedDetected) {
    // Læs flow sensor kontinuerligt
    if (flowSensor_hasPulse()) {
      lastFlowDetectedTime = millis();  // Opdater sidste gang flow saes
    }
    
    // Hvis ingen flow i 2 sekunder = vand er tilbage i tank
    if (millis() - lastFlowDetectedTime > 2000) {
      flowStoppedDetected = true;
      break;
    }
    
    // Sikkerhed: hvis pumpe køres i > 30 sekunder uden flow = fejl
    if (millis() - pumpStartTime > 30000) {
      LCD.print("PUMP FAILURE - CHECK HOSES");
      pumpeStepper.stop();  // STOP pumpen
      return;  // EXIT - kræver manuel indgreb
    }
    
    pumpeStepper.runSpeed();  // Fortsæt pumpning
    delay(100);
  }
  
  // 4) PUMPE STOP (flow er stoppet = vand tilbage)
  pumpeStepper.setSpeed(0);
  pumpeStepper.stop();
  
  // 5) RESET ALLE MOTORER TIL HOME POSITION
  resetMotorsToHome();
  
  // 6) HALT - VENT PÅ BRUGER ACTION
  systemActive = false;
  faultFlag = true;  // Hold orange LED blinkende
}

void resetMotorsToHome() {
  // SOLAR ROTATION (SR1) - TIL 0°
  while (solarRotationPosition != 0) {
    solarXStepper.setSpeed(600);  // Hurtig return
    solarXStepper.setDirection(calculateHomeDirection_SR1());
    solarXStepper.runSpeed();
  }
  solarXStepper.stop();
  
  // SOLAR TILT (ST2) - TIL 45°
  while (solarTiltPosition != 45) {
    solarYStepper.setSpeed(600);
    solarYStepper.setDirection(TILT_DOWN);
    solarYStepper.runSpeed();
  }
  solarYStepper.stop();
  
  // ALGE FRAME TILT (Motor 3) - TIL 45°
  while (algeTiltPosition != 45) {
    algeStepper.setSpeed(600);
    algeStepper.setDirection(TILT_DOWN);
    algeStepper.runSpeed();
  }
  algeStepper.stop();
  
  // Alle motorer er nu i sikker HOME position
}
```
